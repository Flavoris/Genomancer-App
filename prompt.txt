Implement Sequential Fine-Tuning (For 90%+ Accuracy)

To reach msBERT-level accuracy (96.2%), we need to implement sequential fine-tuning where Stage 2 models are initialized from Stage 1 checkpoints.

### Task 1: Create Sequential Training Script

Create a new file `train_sequential.py` that implements the msBERT two-stage approach:
```python
"""
Sequential Fine-Tuning for Gene Whisperer
Implements msBERT's two-stage approach:
1. Fine-tune on promoter identification (Stage 1)
2. Initialize Stage 2 from Stage 1 and fine-tune on strength
"""

import torch
import yaml
from pathlib import Path

def sequential_train(config_path: str):
    """Execute sequential fine-tuning."""
    
    with open(config_path) as f:
        cfg = yaml.safe_load(f)
    
    kmers = cfg.get("multi_scale_kmers", [3, 4, 5, 6])
    
    print("=" * 60)
    print("SEQUENTIAL FINE-TUNING")
    print("=" * 60)
    
    # Phase 1: Train Stage 1 models for each k-mer
    stage1_checkpoints = {}
    for kmer in kmers:
        print(f"\n[Phase 1] Training Stage 1 for k={kmer}")
        print("-" * 40)
        
        # Update config for this k-mer
        cfg_k = dict(cfg)
        cfg_k["kmer"] = kmer
        
        # Train Stage 1
        from train_stage1 import train_stage1_single
        results = train_stage1_single(cfg_k, kmer=kmer)
        
        stage1_checkpoints[kmer] = results["best_ckpt_path"]
        print(f"Stage 1 k={kmer}: Acc={results['best_val_acc']:.4f}, MCC={results['best_val_mcc']:.4f}")
    
    # Phase 2: Sequential fine-tuning for Stage 2
    # (Only if you have Stage 2 data for strength prediction)
    if cfg.get("stage2_train"):
        print(f"\n[Phase 2] Sequential Stage 2 Training")
        print("-" * 40)
        
        for kmer in kmers:
            print(f"\nInitializing Stage 2 k={kmer} from Stage 1 checkpoint")
            
            # Load Stage 1 weights into Stage 2 model
            stage1_ckpt = stage1_checkpoints[kmer]
            
            from train_stage2 import train_stage2_from_stage1
            results = train_stage2_from_stage1(
                cfg=cfg,
                kmer=kmer,
                stage1_checkpoint=stage1_ckpt,
            )
            
            print(f"Stage 2 k={kmer}: Acc={results['best_val_acc']:.4f}")
    
    # Phase 3: Ensemble evaluation
    print(f"\n[Phase 3] Ensemble Evaluation")
    print("-" * 40)
    
    # Load all Stage 1 models and evaluate ensemble
    from evaluate_ensemble import evaluate_soft_voting_ensemble
    ensemble_results = evaluate_soft_voting_ensemble(
        config_path=config_path,
        checkpoint_paths=stage1_checkpoints,
    )
    
    print(f"\nFinal Ensemble Results:")
    print(f"  Accuracy: {ensemble_results['accuracy']:.4f} ({ensemble_results['accuracy']*100:.2f}%)")
    print(f"  MCC: {ensemble_results['mcc']:.4f}")
    print(f"  AUC: {ensemble_results['auc']:.4f}")
    
    return ensemble_results

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--config", default="config.yaml")
    args = parser.parse_args()
    
    sequential_train(args.config)
```

### Task 3: Create Ensemble Evaluation Script

Create `evaluate_ensemble.py`:
```python
"""Evaluate soft voting ensemble on validation set."""

import torch
import yaml
from pathlib import Path
from sklearn.metrics import accuracy_score, matthews_corrcoef, roc_auc_score

def evaluate_soft_voting_ensemble(config_path: str, checkpoint_paths: dict) -> dict:
    """
    Evaluate soft voting ensemble.
    
    Args:
        config_path: Path to config.yaml
        checkpoint_paths: Dict mapping k-mer to checkpoint path
    """
    with open(config_path) as f:
        cfg = yaml.safe_load(f)
    
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    
    # Load models
    models = {}
    for kmer, ckpt_path in checkpoint_paths.items():
        from model import GeneWhispererStage1
        from dataset import KmerVocabulary
        
        # Load vocab
        vocab_dir = Path(cfg.get("vocab_cache_dir", "../artifacts/vocabs"))
        vocab = KmerVocabulary.load(vocab_dir / f"k{kmer}_vocab.json")
        
        # Create model
        model = GeneWhispererStage1(
            vocab_size=len(vocab.itos),
            kmer=kmer,
            embedding_dim=cfg.get("embedding_dim", 384),
            num_layers=cfg.get("transformer_layers", 12),
            num_heads=cfg.get("transformer_heads", 12),
            ff_dim=cfg.get("transformer_ff_dim", 1536),
            dropout=cfg.get("transformer_dropout", 0.1),
            pad_token_id=vocab.pad_id,
            # ... other params
        )
        
        # Load weights
        checkpoint = torch.load(ckpt_path, map_location=device)
        if "model_state_dict" in checkpoint:
            model.load_state_dict(checkpoint["model_state_dict"])
        else:
            model.load_state_dict(checkpoint)
        
        model = model.to(device)
        model.eval()
        models[kmer] = (model, vocab)
    
    # Load validation data
    from dataset import build_dataloaders
    dataloaders = build_dataloaders(cfg)
    val_loader = dataloaders["stage1"]["val"]
    
    # Get raw sequences for multi-kmer tokenization
    val_dataset = val_loader.dataset
    
    all_probs = []
    all_labels = []
    
    with torch.no_grad():
        for idx in range(len(val_dataset)):
            # Get sample
            seq = val_dataset.sequences[idx]
            label = val_dataset.labels[idx]
            
            # Get predictions from each model
            kmer_probs = []
            for kmer, (model, vocab) in models.items():
                tokens = vocab.tokenize(seq, cfg.get("max_bp_len", 81))
                tokens = tokens.unsqueeze(0).to(device)
                
                # Get engineered features
                from dataset import compute_all_features
                eng = compute_all_features(seq).unsqueeze(0).to(device)
                
                logits = model(tokens, eng)
                prob = torch.sigmoid(logits)
                kmer_probs.append(prob.item())
            
            # Soft voting (average probabilities)
            ensemble_prob = sum(kmer_probs) / len(kmer_probs)
            all_probs.append(ensemble_prob)
            all_labels.append(label)
    
    # Compute metrics
    all_probs = torch.tensor(all_probs)
    all_labels = torch.tensor(all_labels)
    
    predictions = (all_probs >= 0.5).float()
    
    accuracy = accuracy_score(all_labels, predictions)
    mcc = matthews_corrcoef(all_labels, predictions)
    auc = roc_auc_score(all_labels, all_probs)
    
    return {
        "accuracy": accuracy,
        "mcc": mcc,
        "auc": auc,
    }
```
